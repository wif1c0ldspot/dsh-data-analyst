Query
WITH bounds AS (
  SELECT date_trunc('day', TIMESTAMP '1970-01-01 00:00:00' + "timestamp" * INTERVAL '1 second') AS day,
         max("timestamp") AS last_ts
  FROM "btcusd_1_min_data"
  GROUP BY day
), daily AS (
  SELECT b.day AS day, t."close" AS day_close
  FROM bounds b
  JOIN "btcusd_1_min_data" t ON t."timestamp" = b.last_ts
), rets AS (
  SELECT 100.0 * ((cur.day_close / nullif(prev.day_close, 0)) - 1) AS daily_return_pct
  FROM daily cur
  JOIN daily prev ON prev.day = cur.day - INTERVAL '1 day'
  WHERE cur.day >= TIMESTAMP '2020-01-01 00:00:00'
)
SELECT floor(daily_return_pct) AS bucket_lower_pct,
       count(*) AS days
FROM rets
GROUP BY bucket_lower_pct
ORDER BY bucket_lower_pct
